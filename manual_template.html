from fastapi import FastAPI
from fastapi.responses import StreamingResponse
import asyncio, json

app = FastAPI(title="Agent Streaming Service")

async def agent_trace_stream(user_query: str):
    stages = [
        ("thought", "正在分析用户意图并提取关键参数..."),
        ("tool_call", "触发外部工具: query_inventory(sku_id='A102')"),
        ("observation", "工具返回: 华东一号仓尚有库存 42 件"),
        ("answer", "为您查询到该商品目前库存充足，可直接下单。")
    ]
    for stage_type, text in stages:
        chunk = json.dumps({"type": stage_type, "content": text}, ensure_ascii=False)
        yield f"data: {chunk}\n\n"
        await asyncio.sleep(0.5)

@app.get("/api/chat/stream")
async def chat_stream(query: str):
    return StreamingResponse(agent_trace_stream(query), media_type="text/event-stream")

原生 Python 实现 ReAct 运行循环

import json

def run_react_loop(llm_client, user_query: str, tools_registry: dict, max_steps: int = 5):
    # 原生实现的 ReAct 运行循环，不依赖任何三方重型编排框架
    messages = [
        {"role": "system", "content": "你是一个能够调用工具解决问题的智能体。在调用工具前请输出思考过程。"},
        {"role": "user", "content": user_query}
    ]

    for step in range(max_steps):
        # 1. 触发大模型推理
        response = llm_client.chat.completions.create(
            model="gpt-4o",
            messages=messages,
            tools=[tool["schema"] for tool in tools_registry.values()]
        )
        msg = response.choices[0].message
        messages.append(msg)

        # 2. 如果模型直接输出文本而不需要调用工具，则任务完成
        if not msg.tool_calls:
            return msg.content

        # 3. 遍历并执行模型决定的工具调用
        for tool_call in msg.tool_calls:
            fn_name = tool_call.function.name
            fn_args = json.loads(tool_call.function.arguments)
            
            tool_func = tools_registry[fn_name]["func"]
            tool_output = tool_func(**fn_args)

            # 4. 将观察结果作为 tool 消息追加回上下文
            messages.append({
                "role": "tool",
                "tool_call_id": tool_call.id,
                "name": fn_name,
                "content": json.dumps(tool_output, ensure_ascii=False)
            })

    return "达到最大思考步数限制，转入人工兜底流程。"

在大型企业系统中，如果每一次用户寒暄都请求 70B 大模型，将造成极大的算力浪费。工业界成熟做法是在网关层使用轻量分类器（如 BERT）在 15ms 内完成分流：

图 4-2：智能客服核心业务需求对照与 BERT 分类模型结构

PyTorch 实现 BERT 意图分类器

import torch
import torch.nn as nn
from transformers import BertModel

class GatewayIntentClassifier(nn.Module):
    # 网关前置意图分类模型 (咨询 / 查单 / 投诉 / 闲聊)
    # 可在 15ms 内过滤 60% 以上高频常见请求
    def __init__(self, bert_path: str, num_classes: int = 4):
        super().__init__()
        self.bert = BertModel.from_pretrained(bert_path)
        self.dropout = nn.Dropout(0.1)
        self.fc = nn.Linear(self.bert.config.hidden_size, num_classes)

    def forward(self, input_ids, attention_mask):
        outputs = self.bert(input_ids=input_ids, attention_mask=attention_mask)
        # 提取 [CLS] 特征进行下游分类
        cls_rep = outputs.pooler_output
        logits = self.fc(self.dropout(cls_rep))
        return logits

6.2 LangGraph 状态图与条件边完整工程代码

LangGraph 通过有向状态图支持循环迭代、工具挂载与死循环中断：

from typing import TypedDict, Annotated, Sequence
import operator
from langchain_core.messages import BaseMessage, HumanMessage, ToolMessage
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool
from langgraph.graph import StateGraph, END

# 1. 定义业务工具
@tool
def check_order_delivery(order_id: str) -> str:
    """根据订单号查询真实物流配送状态"""
    return f"订单 {order_id} 已从华东中转站发出，预计今日 18:00 送达。"

tools = [check_order_delivery]
model = ChatOpenAI(model="gpt-4o", temperature=0).bind_tools(tools)

# 2. 定义全图共享状态
class GraphState(TypedDict):
    messages: Annotated[Sequence[BaseMessage], operator.add]

# 3. 定义图节点
def agent_node(state: GraphState):
    return {"messages": [model.invoke(state["messages"])]}

def tool_node(state: GraphState):
    last_msg = state["messages"][-1]
    results = []
    for call in last_msg.tool_calls:
        output = check_order_delivery.invoke(call["args"])
        results.append(ToolMessage(tool_call_id=call["id"], content=str(output)))
    return {"messages": results}

# 4. 动态路由判断
def router_edge(state: GraphState):
    last_msg = state["messages"][-1]
    if hasattr(last_msg, "tool_calls") and len(last_msg.tool_calls) > 0:
        return "tools"
    return END

# 5. 编译运行图
wf = StateGraph(GraphState)
wf.add_node("agent", agent_node)
wf.add_node("tools", tool_node)
wf.set_entry_point("agent")
wf.add_conditional_edges("agent", router_edge, {"tools": "tools", END: END})
wf.add_edge("tools", "agent")
graph_app = wf.compile()

患者画像动态加权消歧代码实现

from pydantic import BaseModel
from typing import Literal

class PatientRecord(BaseModel):
    age: int
    gender: Literal["male", "female"]
    history: list[str] = []

def clinical_triage_disambiguate(symptom: str, patient: PatientRecord) -> dict:
    """三级消歧：当主诉模糊（胸闷、心慌）时，结合画像动态推理危急等级"""
    if "胸闷" in symptom or "心慌" in symptom:
        if patient.age >= 60 or "高血压" in patient.history:
            return {
                "dept": "急诊科 / 心内科",
                "level": "CRITICAL",
                "guide": "检测到高龄及既往心血管风险，建议立即前往急诊查心电图与心肌标志物！"
            }
        else:
            return {
                "dept": "普通内科 / 心理科",
                "level": "NORMAL",
                "guide": "年轻群体多见于神经衰弱或过度疲劳，建议补充作息情况并预约普通门诊。"
            }
    return {"dept": "全科导诊", "level": "LOW", "guide": "请补充具体的症状持续时间。"}

8.1 生产级防死循环熔断器（Circuit Breaker）实操实现

Agent 如果因工具入参错误而反复重试，会迅速耗尽上下文并造成严重成本事故。必须在执行层注入死循环熔断器：

class AgentCircuitBreaker:
    def __init__(self, max_turns: int = 6, max_repeat_calls: int = 2):
        self.max_turns = max_turns
        self.max_repeat_calls = max_repeat_calls
        self.call_history = []

    def verify_and_record(self, tool_name: str, tool_args: dict):
        self.call_history.append((tool_name, tool_args))
        # 1. 检查总迭代轮次
        if len(self.call_history) > self.max_turns:
            raise RuntimeError("触发最大运行轮次安全熔断，强制中止执行并转人工坐席。")

        # 2. 检查连续相同工具调用
        if len(self.call_history) >= self.max_repeat_calls:
            recent = [name for name, _ in self.call_history[-self.max_repeat_calls:]]
            if len(set(recent)) == 1:
                raise RuntimeError(f"检测到工具 {tool_name} 发生无进展连续重复调用，触发安全拦截。")

8.3 大厂技术面试高频通关真题

Q: 为什么不用超长 Prompt 一次性搞定，而要做节点化工作流？
A: 1. 注意力衰减（Lost in the Middle）与幻觉率随 Prompt 长度指数级上升；2. 无法进行模块化单元测试与灰度发布；3. 无法按节点精细化分配模型算力与混合编排，导致成本与延迟居高不下。

Q: 向量检索在 RAG 场景中遇到生僻词失效怎么办？
A: 必须引入混合检索（Hybrid Search，向量 + BM25 稀疏检索），并结合交叉编码器模型（如 BGE-Reranker）进行二次重排序打分，兼顾语义联想与词法精确匹配。
